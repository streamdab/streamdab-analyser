cmake_minimum_required(VERSION 3.16)
project(StreamDAB_Enhanced VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent Network)
qt6_standard_project_setup()

add_executable(StreamDAB_Enhanced
    main_network_enhanced.cpp
    src/utils/logger.cpp
)

target_include_directories(StreamDAB_Enhanced PRIVATE src)
target_link_libraries(StreamDAB_Enhanced Qt6::Core Qt6::Widgets Qt6::Concurrent Qt6::Network)
