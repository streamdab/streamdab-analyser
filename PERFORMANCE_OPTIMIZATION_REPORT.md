```cpp
int32_t SimdSyncDetector::detect_sync_pattern_avx2(const uint8_t* data, size_t length) const {
    const __m256i pattern = _mm256_load_si256(reinterpret_cast<const __m256i*>(sync_pattern_));
    
    // Process 32-byte chunks with AVX2
    for (size_t i = 0; i <= length - 32; i += 32) {
        const __m256i chunk = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(data + i));
        const __m256i cmp = _mm256_cmpeq_epi32(chunk, pattern);
        
        const uint32_t mask = _mm256_movemask_epi8(cmp);
        if (mask != 0) {
            return static_cast<int32_t>(i); // Found sync pattern
        }
    }
    return -1;
}
```

void SimdAudioProcessor::convert_int16_to_float_avx2(
    const int16_t* input, float* output, size_t sample_count) const {
    
    // Process 16 samples at once with AVX2
    for (size_t i = 0; i < sample_count; i += 16) {
        __m256i int16_samples = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(input + i));
        __m256 float_samples = _mm256_cvtepi32_ps(_mm256_cvtepi16_epi32(int16_samples));
        _mm256_storeu_ps(output + i, float_samples);
    }
}
