struct FIG0_13_UserApp {
    uint32_t service_id;              // 16-bit or 32-bit
    uint8_t component_id;             // SCIdS (4 bits)
    uint16_t user_app_type;           // 11-bit application type
    std::vector<uint8_t> user_app_data;  // Variable length
};

struct FIG0_14_FECScheme {
    uint8_t subchannel_id;            // 6 bits
    uint8_t fec_scheme;               // 2 bits (0=No FEC, 1=RS)
};

struct FIG0_17_ProgrammeType {
    uint16_t service_id;              // Programme service only
    bool is_static;                   // SD flag
    uint8_t programme_type;           // PTy (5 bits, 0-31)
};

struct FIG0_18_AnnouncementSupport {
    uint16_t service_id;
    uint16_t announcement_support_flags;  // 16-bit ASu
    std::vector<uint8_t> cluster_ids;     // Cluster list
};

struct FIG0_19_AnnouncementSwitching {
    uint8_t cluster_id;
    uint16_t announcement_flags;      // 16-bit ASw (ACTIVE)
    uint8_t subchannel_id;
    bool new_flag;                    // New announcement
    bool region_flag;
    uint8_t region_id;                // Optional
};
