cmake_minimum_required(VERSION 3.16)
project(StreamDAB_Enhanced VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Find Qt6 with better error reporting
find_package(Qt6 QUIET COMPONENTS Core Widgets Concurrent Network Gui PrintSupport)

if(Qt6_FOUND)
    message(STATUS "Qt6 found - Core: ${Qt6Core_VERSION}, Widgets: ${Qt6Widgets_VERSION}")
    qt6_standard_project_setup()
else()
    message(FATAL_ERROR "Qt6 not found. Please install Qt6 development packages:
    Ubuntu/Debian: sudo apt install qt6-base-dev qt6-base-dev-tools
    Available Qt6 components: ${Qt6_COMPONENTS_FOUND}")
endif()

# Test with minimal main.cpp
add_executable(StreamDAB_Enhanced
    src/main.cpp
)

target_link_libraries(StreamDAB_Enhanced
    Qt6::Core
    Qt6::Widgets
    Qt6::Concurrent
    Qt6::Network
    Qt6::Gui
    Qt6::PrintSupport
)

# Professional build settings
target_compile_definitions(StreamDAB_Enhanced PRIVATE
    QT_DEPRECATED_WARNINGS
    QT_DISABLE_DEPRECATED_BEFORE=0x060400
)

# Enable all warnings for professional code quality
if(MSVC)
    target_compile_options(StreamDAB_Enhanced PRIVATE /W4)
else()
    target_compile_options(StreamDAB_Enhanced PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Professional deployment configuration
include(GNUInstallDirs)
install(TARGETS StreamDAB_Enhanced
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

# Professional packaging support
include(CPack)
set(CPACK_PACKAGE_NAME "StreamDAB-Analyzer")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Professional ETI Stream Analyzer for Broadcast Industry")
set(CPACK_PACKAGE_VENDOR "StreamDAB Technologies")
set(CPACK_PACKAGE_CONTACT "support@streamdab.com")

# Platform-specific packaging
if(WIN32)
    set(CPACK_GENERATOR "WIX;NSIS")
    set(CPACK_WIX_UPGRADE_GUID "12345678-1234-1234-1234-123456789012")
    set(CPACK_NSIS_DISPLAY_NAME "StreamDAB Analyzer")
elseif(APPLE)
    set(CPACK_GENERATOR "Bundle;DragNDrop")
    set(CPACK_BUNDLE_NAME "StreamDAB Analyzer")
    set(CPACK_BUNDLE_ICON "${CMAKE_SOURCE_DIR}/resources/app.icns")
else()
    set(CPACK_GENERATOR "DEB;RPM;TGZ")

    # Debian/Ubuntu package configuration
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6core6, libqt6widgets6, libqt6gui6, libqt6network6, libqt6concurrent6, libqt6printsupport6")
    set(CPACK_DEBIAN_PACKAGE_SECTION "electronics")
    set(CPACK_DEBIAN_PACKAGE_PRIORITY "optional")

    # RPM package configuration
    set(CPACK_RPM_PACKAGE_REQUIRES "qt6-qtbase >= 6.0.0")
    set(CPACK_RPM_PACKAGE_GROUP "Applications/Engineering")
    set(CPACK_RPM_PACKAGE_LICENSE "Commercial")
endif()
