cmake_minimum_required(VERSION 3.20)
project(StreamDAB_Analyzer_Minimal VERSION 1.1.0 LANGUAGES CXX)

# Modern C++20 standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt6
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

# Enable Qt MOC
set(CMAKE_AUTOMOC ON)

# Create minimal executable
add_executable(StreamDAB_Analyzer_Minimal
    src/main_minimal.cpp
)

# Link Qt6 libraries
target_link_libraries(StreamDAB_Analyzer_Minimal
    Qt6::Core
    Qt6::Widgets
)

# Set target properties
set_target_properties(StreamDAB_Analyzer_Minimal PROPERTIES
    WIN32_EXECUTABLE TRUE
    MACOSX_BUNDLE TRUE
)
