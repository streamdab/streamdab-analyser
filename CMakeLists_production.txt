cmake_minimum_required(VERSION 3.16)
project(StreamDAB_Analyzer_Production VERSION 1.1.0)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Qt6 Configuration
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Gui Network Multimedia)

# Create simple production executable
add_executable(StreamDAB_Analyzer_Production
    main_network_enhanced.cpp
)

target_link_libraries(StreamDAB_Analyzer_Production
    Qt6::Core
    Qt6::Widgets
    Qt6::Gui
    Qt6::Network
    Qt6::Multimedia
)

# Set production optimizations
set_target_properties(StreamDAB_Analyzer_Production PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
)

# Enable Qt MOC
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

message(STATUS "========================================")
message(STATUS "StreamDAB Production Build Configuration")
message(STATUS "========================================")
message(STATUS "Version: ${PROJECT_VERSION}")
message(STATUS "Build Type: Release")
message(STATUS "Qt6 Version: ${Qt6_VERSION}")
message(STATUS "========================================")
