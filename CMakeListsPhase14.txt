# Phase 1.4 Complete DAB-XPlorer Application
cmake_minimum_required(VERSION 3.16)

# Add Phase 1.4 Complete DAB-XPlorer executable
add_executable(Phase14_DAB_XPlorer_Complete
    phase1_4_dab_xplorer_complete.cpp
    src/core/enhanced_eti_processor_qt.cpp
    src/core/advanced_fig_analyzer.cpp
    src/core/eti_types.cpp
)

# Qt6 requirements
target_link_libraries(Phase14_DAB_XPlorer_Complete
    Qt6::Core
    Qt6::Widgets
    Qt6::Gui
)

# Include directories
target_include_directories(Phase14_DAB_XPlorer_Complete PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)

# C++20 standard
set_target_properties(Phase14_DAB_XPlorer_Complete PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    AUTOMOC ON
)

# Copy ETI file to build directory for testing
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/eti/bkk_20062022_141637.eti
    ${CMAKE_CURRENT_BINARY_DIR}/eti/bkk_20062022_141637.eti
    COPYONLY
)
