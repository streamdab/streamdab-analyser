cmake_minimum_required(VERSION 3.16)
project(StreamDAB_Enhanced VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Professional build type configuration
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Release" CACHE STRING "Choose the type of build." FORCE)
    set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS "Debug" "Release" "RelWithDebInfo" "MinSizeRel")
endif()

# Qt6 components
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent Network Gui PrintSupport)
qt6_standard_project_setup()

# Use minimal main for deployment validation
add_executable(StreamDAB_Enhanced
    src/main_minimal.cpp
)

# Include directories
target_include_directories(StreamDAB_Enhanced PRIVATE
    src
)

# Link Qt6 libraries
target_link_libraries(StreamDAB_Enhanced
    Qt6::Core
    Qt6::Widgets
    Qt6::Concurrent
    Qt6::Network
    Qt6::Gui
    Qt6::PrintSupport
)

# Professional build settings
target_compile_definitions(StreamDAB_Enhanced PRIVATE
    QT_DEPRECATED_WARNINGS
    QT_DISABLE_DEPRECATED_BEFORE=0x060400
    STREAMDAB_VERSION="${PROJECT_VERSION}"
)

# Compiler-specific settings
if(MSVC)
    target_compile_options(StreamDAB_Enhanced PRIVATE /W4)
    target_compile_definitions(StreamDAB_Enhanced PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(StreamDAB_Enhanced PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Professional installation configuration
include(GNUInstallDirs)

# Install the executable
install(TARGETS StreamDAB_Enhanced
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

# Install documentation if it exists
if(EXISTS "${CMAKE_SOURCE_DIR}/README.md")
    install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
endif()

# Basic package information - MUST be set BEFORE include(CPack)
set(CPACK_PACKAGE_NAME "StreamDAB-Analyzer")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Professional ETI Stream Analyzer for Broadcast Industry")
set(CPACK_PACKAGE_DESCRIPTION "Advanced ETSI-compliant ETI stream analysis tool for professional broadcast monitoring and DAB+ service discovery with real-time processing capabilities.")
set(CPACK_PACKAGE_VENDOR "StreamDAB Technologies")
set(CPACK_PACKAGE_CONTACT "support@streamdab.com")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://streamdab.com")

# Required for all package formats
set(CPACK_PACKAGE_MAINTAINER "StreamDAB Technologies <support@streamdab.com>")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "StreamDAB Technologies <support@streamdab.com>")

# Platform-specific packaging
if(WIN32)
    # Windows MSI and NSIS installers
    set(CPACK_GENERATOR "WIX;NSIS")

    # WIX (MSI) configuration
    set(CPACK_WIX_UPGRADE_GUID "12345678-1234-1234-1234-123456789012")
    set(CPACK_WIX_PRODUCT_GUID "87654321-4321-4321-4321-210987654321")
    set(CPACK_WIX_PROGRAM_MENU_FOLDER "StreamDAB Analyzer")
    set(CPACK_WIX_PROPERTY_ARPCONTACT "support@streamdab.com")
    set(CPACK_WIX_PROPERTY_ARPURLINFOABOUT "https://streamdab.com")

    # NSIS configuration
    set(CPACK_NSIS_DISPLAY_NAME "StreamDAB Analyzer")
    set(CPACK_NSIS_PACKAGE_NAME "StreamDAB Analyzer")
    set(CPACK_NSIS_MENU_LINKS "bin/StreamDAB_Enhanced" "StreamDAB Analyzer")

elseif(APPLE)
    # macOS Bundle and DMG
    set(CPACK_GENERATOR "Bundle;DragNDrop")

    # Bundle configuration
    set(CPACK_BUNDLE_NAME "StreamDAB Analyzer")

    # DMG configuration
    set(CPACK_DMG_VOLUME_NAME "StreamDAB Analyzer")
    set(CPACK_DMG_FORMAT "UDBZ")

else()
    # Linux DEB and RPM packages
    set(CPACK_GENERATOR "DEB;RPM;TGZ")

    # Debian/Ubuntu package configuration
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6core6 (>= 6.0.0), libqt6widgets6 (>= 6.0.0), libqt6gui6 (>= 6.0.0), libqt6network6 (>= 6.0.0), libqt6concurrent6 (>= 6.0.0), libqt6printsupport6 (>= 6.0.0)")
    set(CPACK_DEBIAN_PACKAGE_SECTION "electronics")
    set(CPACK_DEBIAN_PACKAGE_PRIORITY "optional")
    set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://streamdab.com")
    set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/debian/postinst;${CMAKE_CURRENT_SOURCE_DIR}/packaging/debian/prerm")

    # RPM package configuration
    set(CPACK_RPM_PACKAGE_REQUIRES "qt6-qtbase >= 6.0.0")
    set(CPACK_RPM_PACKAGE_GROUP "Applications/Engineering")
    set(CPACK_RPM_PACKAGE_LICENSE "Commercial")
    set(CPACK_RPM_PACKAGE_VENDOR "StreamDAB Technologies")
    set(CPACK_RPM_PACKAGE_URL "https://streamdab.com")

    # Desktop entry for Linux
    set(DESKTOP_ENTRY "[Desktop Entry]
Version=1.0
Type=Application
Name=StreamDAB Analyzer
Comment=Professional ETI Stream Analyzer
Exec=${CMAKE_INSTALL_PREFIX}/bin/StreamDAB_Enhanced
Icon=streamdab-analyzer
Terminal=false
Categories=AudioVideo;Audio;Engineering;
")

    file(WRITE "${CMAKE_BINARY_DIR}/streamdab-analyzer.desktop" "${DESKTOP_ENTRY}")
    install(FILES "${CMAKE_BINARY_DIR}/streamdab-analyzer.desktop"
        DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()

# Professional packaging configuration - MUST be AFTER all CPACK_* variables
include(CPack)

# Professional deployment summary
message(STATUS "========================================")
message(STATUS "StreamDAB Analyzer - Professional Build Configuration")
message(STATUS "========================================")
message(STATUS "Version: ${PROJECT_VERSION}")
message(STATUS "Build Type: ${CMAKE_BUILD_TYPE}")
message(STATUS "Qt6 Version: ${Qt6_VERSION}")
message(STATUS "Install Prefix: ${CMAKE_INSTALL_PREFIX}")
message(STATUS "Available Generators: ${CPACK_GENERATOR}")
message(STATUS "========================================")
