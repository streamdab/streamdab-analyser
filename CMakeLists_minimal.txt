cmake_minimum_required(VERSION 3.16)
project(ETIStreamAnalyzer_Minimal VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt6 with only essential components
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
qt6_standard_project_setup()

# Create minimal executable with core files only
add_executable(ETIStreamAnalyzer_Minimal
    src/main.cpp
    src/utils/logger.cpp
    src/core/eti_processor.cpp
)

target_include_directories(ETIStreamAnalyzer_Minimal PRIVATE src)
target_link_libraries(ETIStreamAnalyzer_Minimal
    Qt6::Core
    Qt6::Widgets
    Qt6::Network
)
