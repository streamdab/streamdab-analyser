cmake_minimum_required(VERSION 3.16)
project(NetworkValidationTest)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt6
find_package(Qt6 REQUIRED COMPONENTS Core Network)

# Add include directories
include_directories(${CMAKE_CURRENT_SOURCE_DIR})

# Add executable
add_executable(network_validation_test test_network_validation.cpp)

# Link Qt libraries
target_link_libraries(network_validation_test Qt6::Core Qt6::Network)

# Enable automatic MOC
set_target_properties(network_validation_test PROPERTIES AUTOMOC ON)
