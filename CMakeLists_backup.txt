cmake_minimum_required(VERSION 3.21)
project(StreamDAB_Phase1_4 VERSION 1.4.0 LANGUAGES CXX)

# ========================================
# StreamDAB Phase 1.4 - Minimal Build for Qt MOC Testing
# ========================================
message(STATUS "========================================")
message(STATUS "StreamDAB Phase 1.4 - Qt MOC Minimal Build")
message(STATUS "========================================")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Qt6 components - Essential only
find_package(Qt6 REQUIRED COMPONENTS
    Core
    Widgets
    Network
)
qt6_standard_project_setup()

message(STATUS "Qt6 Version: ${Qt6_VERSION}")

# Essential libraries for ETI processing
find_package(PkgConfig REQUIRED)
pkg_check_modules(FAAD2 REQUIRED faad2)
pkg_check_modules(FFTW3 REQUIRED fftw3)

# ========================================
# PHASE 1.4 MINIMAL SOURCES
# ========================================

# Core Phase 1.4 sources - only what exists and is needed
set(PHASE1_4_SOURCES
    # Phase 1.2-1.4 Qt Integration (CRITICAL for MOC generation)
    src/core/enhanced_eti_processor_qt.cpp
    src/core/advanced_fig_analyzer.cpp

    # Essential ETI types and processing (verified existing)
    src/core/eti_types.cpp
    src/core/eti_processor.cpp
    src/core/eti_file_worker.cpp

    # Basic GUI for testing
    src/gui/main_window.cpp

    # Entry point
    src/main.cpp
)

# ========================================
# PHASE 1.4 MINIMAL EXECUTABLE
# ========================================

add_executable(StreamDAB_Phase1_4
    ${PHASE1_4_SOURCES}
)

# Include directories
target_include_directories(StreamDAB_Phase1_4 PRIVATE
    src
    src/core
    src/gui
    ${FAAD2_INCLUDE_DIRS}
    ${FFTW3_INCLUDE_DIRS}
)

# Link Qt6 libraries - Essential only
target_link_libraries(StreamDAB_Phase1_4
    Qt6::Core
    Qt6::Widgets
    Qt6::Network
    ${FAAD2_LIBRARIES}
    ${FFTW3_LIBRARIES}
)

# Professional build settings
target_compile_definitions(StreamDAB_Phase1_4 PRIVATE
    QT_DEPRECATED_WARNINGS
    QT_DISABLE_DEPRECATED_BEFORE=0x060400
    STREAMDAB_VERSION="${PROJECT_VERSION}"
    PHASE_1_4_BUILD=1
)

# Add pkg-config settings
target_include_directories(StreamDAB_Phase1_4 PRIVATE
    ${FAAD2_INCLUDE_DIRS}
    ${FFTW3_INCLUDE_DIRS}
)

target_compile_options(StreamDAB_Phase1_4 PRIVATE
    ${FAAD2_CFLAGS_OTHER}
    ${FFTW3_CFLAGS_OTHER}
)

# Compiler settings
if(MSVC)
    target_compile_options(StreamDAB_Phase1_4 PRIVATE /W4)
else()
    target_compile_options(StreamDAB_Phase1_4 PRIVATE -Wall -Wextra)
endif()

message(STATUS "Phase 1.4 Minimal Build Configuration Complete")
message(STATUS "Target: StreamDAB_Phase1_4")
message(STATUS "Source Count: ${PHASE1_4_SOURCES}")
message(STATUS "========================================")
