cmake_minimum_required(VERSION 3.16)
project(StreamDAB_Core_Test VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent)
qt6_standard_project_setup()

add_executable(StreamDAB_Core_Test
    main_core_test.cpp
    src/utils/logger.cpp
)

target_include_directories(StreamDAB_Core_Test PRIVATE src)
target_link_libraries(StreamDAB_Core_Test Qt6::Core Qt6::Widgets Qt6::Concurrent)
